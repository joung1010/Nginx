### Certbot이 작성한 HTTPS 관련 코드 분석

```bash
### Certbot이 작성한 HTTPS 관련 코드 상세 분석

# ===== HTTPS 서버 블록 (443 포트) =====
**# server 블록에서는 일반적으로 listen, server_name을 가장 먼저 처리한다.**
server {
    # 도메인명 매칭: nginx.jhcodeapi.com으로 들어오는 요청만 처리
    server_name nginx.jhcodeapi.com;

    # React 정적 파일 서빙 설정
    location / {
        root /usr/share/nginx/nginx-basic-react/build;
        index index.html;
        
        # **추가 설명**: SPA(Single Page Application) 라우팅을 위해서는
        # try_files $uri $uri/ /index.html; 설정이 필요할 수 있음
    }

    # **SSL/TLS 설정 (Certbot이 자동 추가)**
    listen 443 ssl; # HTTPS 포트, SSL 활성화
    
    # **인증서 파일**: 공개키 + 중간 인증서 체인
    ssl_certificate /etc/letsencrypt/live/nginx.jhcodeapi.com/fullchain.pem;
    
    # **개인키 파일**: 암호화/복호화용 비밀키 (절대 외부 노출 금지!)
    ssl_certificate_key /etc/letsencrypt/live/nginx.jhcodeapi.com/privkey.pem;
    
    # **SSL 보안 설정**: Let's Encrypt 권장 보안 옵션들
    # (TLS 버전, 암호화 알고리즘, 보안 헤더 등)
    include /etc/letsencrypt/options-ssl-nginx.conf;
    
    # **DH 매개변수**: Perfect Forward Secrecy를 위한 키 교환 보안 강화
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
    # Next.js 애플리케이션 서버 (admin 서브도메인)
    server_name admin.jhcodeapi.com;

    location / {
        root /usr/share/nginx/nginx-basic-nextjs/out;
        index index.html;
    }

    # **동일한 SSL 설정** (admin 도메인용 인증서)
    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/admin.jhcodeapi.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/admin.jhcodeapi.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

# ===== HTTP to HTTPS 리다이렉트 (80 포트) =====

server {
    # **잘못된 설정 - 문법 오류!**
    # if 조건이 listen/server_name보다 먼저 나와서는 안됨
    if ($host = nginx.jhcodeapi.com) {
        return 301 https://$host$request_uri; # 이 줄이 빠져있음
    }
    
    listen 80;
    server_name nginx.jhcodeapi.com;
    return 404; # 이 부분이 실행되지 않을 수 있음
}

server {
    # **올바른 HTTP to HTTPS 리다이렉트 설정**
    listen 80;
    server_name admin.jhcodeapi.com;
    
    # Host 헤더 검증 후 HTTPS로 301 리다이렉트
    if ($host = admin.jhcodeapi.com) {
        return 301 https://$host$request_uri;
    }
    
    # 매칭되지 않은 요청은 404 처리
    return 404;
}
```
